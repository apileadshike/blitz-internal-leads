Send rows to a Google Sheet (one-time setup, about 2 minutes)

  1. Create a new Google Sheet (or open the one you want the leads in).
  2. In the sheet, click Extensions → Apps Script.
  3. Delete everything in the editor and paste the code below, then click Save.
  4. Click Deploy → New deployment. Pick the type Web app (gear icon).
    Set Execute as: Me and Who has access: Anyone. Click Deploy and allow the permissions.
  5. Copy the Web app URL (it ends in /exec) and paste it into the Apps Script URL box of the "Send to Google Sheet" step.

The first row of the sheet becomes the column headers automatically. New columns are added on the right as new kinds of data appear. Anyone with the URL can add rows to this sheet, so keep the URL private.

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(30000); // one write at a time so rows never overwrite each other
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    var row = JSON.parse(e.postData.contents).row;
    var lastCol = sheet.getLastColumn();
    var headers = lastCol ? sheet.getRange(1, 1, 1, lastCol).getValues()[0] : [];
    var missing = Object.keys(row).filter(function (k) { return headers.indexOf(k) === -1; });
    if (missing.length) {
      headers = headers.concat(missing);
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    }
    sheet.appendRow(headers.map(function (h) { return row[h] === undefined ? "" : row[h]; }));
    return ContentService.createTextOutput("ok");
  } finally {
    lock.releaseLock();
  }
}
